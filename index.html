from js import document def generate_sku(): # 1. Handling Input using PyScript and Datatypes category_input = document.getElementById("category").value # String name_input = document.getElementById("pname").value # String qty_input = document.getElementById("quantity").value # String/Integer # Check if inputs are empty if category_input == "" or name_input == "" or qty_input == "": document.getElementById("output").innerText = "Please select all options and enter a quantity!" return # 2. Data as Values - Using split() method # Take the first 3 letters of category (converted to uppercase) cat_code = category_input.upper()[:3] # Split the product name into words to extract initials/first letters words_list = name_input.split(" ") name_letters = [] for word in words_list: if len(word) > 0: name_letters.append(word[0].upper()) # 3. Data as Values - Using join() method # Join the letters of the product name together name_code = "".join(name_letters) # 4. Converting Quantity to Integer type stock_num = int(qty_input) # Construct the final SKU code (e.g., COO-CC-50) sku_parts = [cat_code, name_code, str(stock_num)] final_sku = "-".join(sku_parts) # Output result to HTML document.getElementById("output").innerText = "Generated SKU: " + final_sku
